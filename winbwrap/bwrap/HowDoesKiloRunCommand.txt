	有效的取值范围有

	"pwsh"	true	pwsh.exe -NoLogo -NoProfile -NonInteractive -EncodedCommand <双层Base64>
	"powershell"	true	powershell.exe + 同上参数
	"bash"	false	Node.js spawn 包裹 bash.exe -c "命令"
	"cmd"	false	Node.js spawn 包裹 cmd.exe /d /s /c "命令"

（1）ai 希望执行命令

		AiCommand1 param1 param2 param3; AiCommand2 param4 param5

	pwsh 等效执行

		pwsh.exe -NoLogo -NoProfile -NonInteractive -EncodedCommand <UTF16LE-Base64>

	在这里 -EncodedCommand <UTF16LE-Base64> 是将命令转为了编码，避免一下问题

		$env:PATH	PowerShell 的 $ 变量引用
		"双引号"		与 -Command 参数的外层引号冲突
		` 反引号		PowerShell 转义符
		'单引号'		也需要正确匹配
		&, `		, ;`
		嵌套引号		多层转义让人崩溃

	它不会将上述命令直接变为一下方式执行

		pwsh.exe -NoLogo -NoProfile -Command "AiCommand1 param1 param2 param3; AiCommand2 param4 param5"
